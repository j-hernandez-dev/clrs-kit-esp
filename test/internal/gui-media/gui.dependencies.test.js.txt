const runtime = { executable: process.execPath, version: process.versions.node, abi: process.versions.modules,
    platform: process.platform, arch: process.arch };
const lock = { lockfileVersion: 3, packages: { "": { dependencies: { "node-gtk": "4.1.1" } },
    "node_modules/node-gtk": { version: "4.1.1", integrity: "sha512-fixture" } } };
const failure = code => error => error.code === code;

async function fixture(t) {
    const root = await fs.mkdtemp(path.join(os.tmpdir(), "clrs-gui-dependencies-"));
    t.after(() => fs.rm(root, { recursive: true, force: true }));
    const calls = [];
    const options = { extensionRoot: root, lockOptions: { pollMs: 5, timeoutMs: 2000 },
        inspector: { async inspect() { calls.push("node"); return runtime; } },
        npm: { async install({ packages, locked }) {
            calls.push(locked ? "ci" : "install");
            await fs.writeFile(path.join(packages, "package-lock.json"), JSON.stringify(lock));
        } },
        resolver: { async resolve(packages) { calls.push("resolve"); return { entryUrl: pathToFileURL(path.join(packages, "entry.mjs")).href }; } },
        probe: { async check() { calls.push("probe"); return [4, 22, 4]; } } };
    const manager = new NodeGtkDependencyManager(options);
    return { root, calls, options, manager };
}

test("GUI dependencias: primera instalación transaccional y segunda ejecución sin npm", async t => {
    const { manager, calls, root } = await fixture(t);
    const result = await manager.ensure();
    assert.equal(manager.state, "READY");
    assert.match(result.entryUrl, /\/packages\/entry\.mjs$/u);
    assert.equal(result.nodeExecutable, process.execPath);
    assert.equal(calls.filter(x => x === "install").length, 1);
    assert.equal(calls.filter(x => x === "probe").length, 2, "Se comprueba staging y ubicación final");
    await manager.ensure();
    assert.equal(calls.filter(x => x === "install").length, 1);
    assert.deepEqual(await fs.readdir(path.join(root, ".runtime", "staging")), []);
    assert.deepEqual(await fs.readdir(path.join(root, ".runtime", "locks")), []);
    assert.deepEqual(JSON.parse(await fs.readFile(path.join(root, ".runtime/packages/package.json"))), PRIVATE_GUI_MANIFEST);
});

test("GUI dependencias: concurrencia comparte instalación y cancelar un cliente no cancela los demás", async t => {
    const { manager, options, calls } = await fixture(t);
    const original = options.npm.install;
    let started;
    const installing = new Promise(resolve => { started = resolve; });
    options.npm.install = async request => { started(); await delay(35); return original(request); };
    const controller = new AbortController();
    const first = manager.ensure({ signal: controller.signal });
    const cancelled = assert.rejects(first, failure("CLRS_GUI_DEP_CANCELLED"));
    const second = new NodeGtkDependencyManager(options).ensure();
    await installing; controller.abort();
    await Promise.all([cancelled, second]);
    assert.equal(calls.filter(x => x === "install").length, 1);
});

test("GUI lock: exclusión entre procesos, propietario vivo no caduca y propietario muerto se recupera", async t => {

test("GUI lock: dos procesos Node independientes no entran simultáneamente", async t => {
    const { root } = await fixture(t);
    const layoutUrl = new URL("../src/infrastructure/gui/PrivateRuntimeLayout.js", import.meta.url).href;
    const lockUrl = new URL("../src/infrastructure/gui/PrivateRuntimeLock.js", import.meta.url).href;
    const source = `
        import {PrivateRuntimeLayout} from ${JSON.stringify(layoutUrl)};
        import {acquirePrivateRuntimeLock} from ${JSON.stringify(lockUrl)};
        import {writeFile} from 'node:fs/promises';
        import {setTimeout as delay} from 'node:timers/promises';
        const layout = new PrivateRuntimeLayout(process.argv[1]); await layout.prepare();
        const release = await acquirePrivateRuntimeLock(layout, {pollMs:5,timeoutMs:3000});
        const acquired=Date.now();
        if(process.argv[2]==='first') { await writeFile(process.argv[1]+'/entered',''); await delay(180); }
        const released=Date.now(); await release();
        console.log(JSON.stringify({acquired,released}));
    `;
    const processes = new GuiProcessAdapter();
    const first = processes.run(process.execPath, ["--input-type=module", "-e", source, root, "first"]);
    let entered = false;
    for (let attempt = 0; attempt < 200; attempt++) {
        try { await fs.access(path.join(root, "entered")); entered = true; break; } catch { await delay(5); }
    }
    assert.equal(entered, true);
    const second = processes.run(process.execPath, ["--input-type=module", "-e", source, root, "second"]);
    const [a, b] = await Promise.all([first, second]);
    assert.equal(a.code, 0, a.stderr); assert.equal(b.code, 0, b.stderr);
    assert.ok(JSON.parse(b.stdout).acquired >= JSON.parse(a.stdout).released);
});

test("GUI pipeline: sólo run válido provisiona, nunca análisis/exportación; mismo emisor y runner", async () => {
    const calls = [];
    const preparation = new GuiRuntimePreparation({ dependencyManager: { async ensure() {
        calls.push("ensure"); return { entryUrl: "file:///private/entry.mjs", nodeExecutable: process.execPath, version: "4.1.1" };
    } }, getFactorySource: () => "()=>({invoke(){},waitForIdle(){},dispose(){}})" });
    const service = new CompilationService({ runtimePreparation: preparation,
        fileEmitter: { async writeArtifact() { calls.push("emit"); } },
        programRunner: { async run(request) { calls.push("run"); return request; } } });
    await service.runSource("escribir 1", "p.clrs");
    assert.deepEqual(calls, ["emit", "run"]);
    calls.length = 0;
    await service.tryRunSource('BOTON("b")', "p.clrs");
    await service.tryCompileSource('BOTON("b", "x")', "p.clrs");
    assert.deepEqual(calls, []);
    const result = await service.runSource('BOTON("b", "x")', "p.clrs");
    assert.deepEqual(calls, ["ensure", "emit", "run"]);
    assert.equal(result.execution.command, process.execPath);
    assert.match(result.artifact.outputPath, /\.mjs$/u);
    assert.deepEqual(result.artifact.requiredCapabilities, ["gui"]);
});
