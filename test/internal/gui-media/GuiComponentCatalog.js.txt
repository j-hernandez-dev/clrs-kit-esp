export const GUI_COMPONENTS = freezeGuiContract(Object.fromEntries(
    Object.entries(components).map(([type, component], index) => {
        const widget = component.widget !== false;
        return [type, {
            ...component, type, gtkClass: `Gtk.${type}`, widget,
            source: { document: "docs/gui/api-catalog.md", heading: `9.${index + 1} ${type}` },
            containment: component.containment ?? "none",
            constraints: component.constraints ?? [],
            properties: Object.fromEntries(Object.entries({
                ...(widget ? GUI_WIDGET_PROPERTIES : {}), ...component.properties
            }).map(([name, descriptor]) => [name, { name, ...descriptor,
                read: descriptor.access !== "W", write: descriptor.access !== "R" } ])),
            events: Object.fromEntries(Object.entries({
                ...(widget ? GUI_WIDGET_EVENTS : {}), ...component.events
            }).map(([name, descriptor]) => [name, { name, connectOnDemand: true, ...descriptor }])),
            actions: { ...(widget ? widgetActions : {}), ...component.actions }
        }];
    })
));

export function getGuiComponentDefinition(type) {
    return getGuiOwnEntry(GUI_COMPONENTS, type);
}
